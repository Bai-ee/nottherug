const R='/Users/bballi/Documents/Repos/NotTheRug-014-s2/';
const {chromium}=require(R+'node_modules/playwright');
const fs=require('fs');const axeSrc=fs.readFileSync(R+'node_modules/axe-core/axe.min.js','utf8');
const base='http://localhost:3020';
const routes=['/','/about','/safety','/reviews','/book','/contact','/signup','/walk-with-us','/neighborhoods/williamsburg'];
(async()=>{const b=await chromium.launch();const out={};const con={};
for(const w of [390,1440]){
 for(const r of routes){
  const ctx=await b.newContext({viewport:{width:w,height:w==390?844:900}});const p=await ctx.newPage();
  const logs=[];p.on('console',m=>{if(['error','warning'].includes(m.type()))logs.push(m.type()+': '+m.text().slice(0,200))});p.on('pageerror',e=>logs.push('pageerror: '+String(e).slice(0,200)));
  try{await p.goto(base+r,{waitUntil:'load',timeout:45000});}catch(e){}
  await p.waitForTimeout(5000);
  await p.evaluate(async()=>{for(let y=0;y<document.body.scrollHeight;y+=500){window.scrollTo(0,y);await new Promise(r=>setTimeout(r,120));}window.scrollTo(0,0);});
  await p.waitForTimeout(1500);
  await p.evaluate(s=>{eval(s)},axeSrc);
  const res=await p.evaluate(async()=>await axe.run(document,{runOnly:['wcag2a','wcag2aa','wcag21a','wcag21aa','wcag22aa','best-practice']}));
  out[w+r]=res.violations.map(v=>v.id+' x'+v.nodes.length+' '+v.nodes.slice(0,3).map(n=>n.target.join(' ')).join(' | '));
  con[w+r]=logs;console.log(w,r,'axe',res.violations.length,'console',logs.length);
  await ctx.close();}
 // modal
 const ctx=await b.newContext({viewport:{width:w,height:900}});const p=await ctx.newPage();
 await p.goto(base+'/',{waitUntil:'load'});await p.waitForTimeout(3000);
 if(w===390){await p.click('#nav-hamburger-toggle');await p.click('#mobile-menu-book-cta');}else await p.click('#main-nav .nav-cta');
 await p.waitForSelector('#welcome-walk-modal',{state:'visible'});await p.waitForTimeout(1200);
 await p.evaluate(s=>{eval(s)},axeSrc);
 const res=await p.evaluate(async()=>await axe.run(document,{runOnly:['wcag2a','wcag2aa','wcag21a','wcag21aa','wcag22aa','best-practice']}));
 out[w+' modal']=res.violations.map(v=>v.id+' x'+v.nodes.length+' '+v.nodes.slice(0,3).map(n=>n.target.join(' ')).join(' | '));
 console.log(w,'modal axe',res.violations.length);await ctx.close();
}
fs.writeFileSync(R+'plans/reports/014-S2-evidence/gate/axe-console.json',JSON.stringify({axe:out,console:con},null,1));await b.close();})();
