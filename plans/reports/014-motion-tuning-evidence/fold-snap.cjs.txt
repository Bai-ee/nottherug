const { chromium } = require('/Users/bballi/Documents/Repos/NotTheRug-motion/node_modules/playwright');
const fs = require('fs');
const [, , outDir, base, paths] = process.argv;
(async () => {
  fs.mkdirSync(outDir, { recursive: true });
  const browser = await chromium.launch();
  for (const p of paths.split(',')) {
    for (const [w, h] of [[390, 844], [1440, 900]]) {
      const ctx = await browser.newContext({ viewport: { width: w, height: h }, reducedMotion: 'reduce', deviceScaleFactor: 1 });
      const page = await ctx.newPage();
      await page.goto(base + p, { waitUntil: 'load' });
      await page.addStyleTag({ content: '.proof-track,.trust-bar-track{animation:none !important;transform:none !important}#main-nav{transition:none !important}' });
      await page.waitForTimeout(3000);
      const H = await page.evaluate(() => Math.max(document.body.scrollHeight, document.documentElement.scrollHeight));
      for (let y = 0; y < H; y += 700) { await page.evaluate((v) => window.scrollTo(0, v), y); await page.waitForTimeout(100); }
      await page.evaluate(() => window.scrollTo(0, 0));
      await page.waitForTimeout(1200);
      const slug = (p === '/' ? 'home' : p.replace(/\W+/g, '')) + (process.env.DIALOG ? '-dialog' : '');
      if (process.env.DIALOG && p === '/') {
        await page.evaluate(() => { const e = [...document.querySelectorAll('#home-rates-preview-row button, #home-rates-preview-row a')].find((x) => /discount/i.test(x.textContent)); e.click(); });
        await page.waitForSelector('#service-modal .service-modal-rate-amount', { timeout: 10000 }); await page.waitForTimeout(1500);
      }
      await page.screenshot({ path: `${outDir}/${slug}-${w}.png`, fullPage: !process.env.DIALOG });
      const dump = await page.evaluate(() => {
        const els = [...document.querySelectorAll('body, body *')].filter((e) => !e.closest('nextjs-portal'));
        return { rows: els.map((e) => { const cs = getComputedStyle(e); const path = e.tagName.toLowerCase() + (e.id ? '#' + e.id : '') + (typeof e.className === 'string' && e.className ? '.' + e.className.split(/\s+/).join('.') : ''); const o = {}; for (const q of cs) o[q] = cs.getPropertyValue(q); return [path, o]; }), h: Math.max(document.body.scrollHeight, document.documentElement.scrollHeight) };
      });
      fs.writeFileSync(`${outDir}/${slug}-${w}.json`, JSON.stringify(dump));
      await ctx.close();
    }
  }
  await browser.close();
})();
